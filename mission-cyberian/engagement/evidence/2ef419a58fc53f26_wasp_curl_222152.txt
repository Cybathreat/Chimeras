// Configuration
const postsPerPage = 10;

// Parse URL query params
function getUrlParams() {
  const params = new URLSearchParams(window.location.search);
  let page = parseInt(params.get('page')) || 1;
  if (isNaN(page) || page < 1) page = 1;
  return { page };
}

// Render posts for current page
function renderPosts() {
  const { page } = getUrlParams();
  const totalPages = Math.ceil(blogPosts.length / postsPerPage);
  const start = (page - 1) * postsPerPage;
  const currentPosts = blogPosts.slice(start, start + postsPerPage);

  const container = document.getElementById('blog-posts-container');

  if (currentPosts.length === 0) {
    container.innerHTML = '<div class="text-center py-10"><p class="text-text-secondary">No articles found.</p></div>';
    return;
  }

  container.innerHTML = currentPosts.map(post => `
    <article class="bg-surface border border-border p-6">
      <div class="flex items-center gap-3 mb-4">
        <span class="font-mono text-xs text-navy bg-navy-dim px-2 py-0.5">${post.tags.join(' ')}</span>
        <span class="font-mono text-xs text-text-muted">${post.date}</span>
      </div>
      <h2 class="text-2xl font-bold mb-3">
        <a href="${post.url}" class="text-text-primary hover:text-navy transition-colors">
          ${post.title}
        </a>
      </h2>
      <p class="text-text-secondary mb-4">
        ${post.excerpt}
      </p>
      <div class="flex items-center justify-between">
        <span class="text-text-secondary text-sm">By Ahmed Chiboub</span>
        <a href="${post.url}" class="text-navy hover:text-text-primary transition-colors text-sm font-semibold">
          Read More
        </a>
      </div>
    </article>
  `).join('\n');

  // Show pagination only if more than 10 posts
  if (blogPosts.length > 10) {
    renderPagination(page, totalPages);
  }
}

// Render pagination controls
function renderPagination(currentPage, totalPages) {
  const controls = document.getElementById('pagination-controls');
  controls.classList.remove('hidden');

  // Update summary
  document.getElementById('pagination-summary').textContent = `Page ${currentPage} of ${totalPages} (${blogPosts.length} posts total)`;

  // Generate page links
  const pagesContainer = document.getElementById('pagination-pages');
  let pageLinks = '';

  // Calculate page numbers to show (max 5 pages visible)
  let startPage = Math.max(1, currentPage - 2);
  let endPage = Math.min(totalPages, startPage + 4);
  if (endPage - startPage < 4) {
    startPage = Math.max(1, endPage - 4);
  }

  for (let p = startPage; p <= endPage; p++) {
    if (p === currentPage) {
      pageLinks += `<span class="inline-flex items-center justify-center w-10 h-10 font-semibold text-text-primary">${p}</span>`;
    } else {
      const url = p === 1 ? '/blog' : `/blog?page=${p}`;
      pageLinks += `<a href="${url}" class="inline-flex items-center justify-center w-10 h-10 font-semibold text-text-secondary hover:text-text-primary transition-colors">${p}</a>`;
    }
  }
  pagesContainer.innerHTML = pageLinks;

  // Previous/Next buttons
  const prevBtn = document.getElementById('pagination-prev');
  const nextBtn = document.getElementById('pagination-next');

  if (currentPage > 1) {
    prevBtn.innerHTML = `<a href="/blog?page=${currentPage - 1}" class="text-sm font-semibold text-text-secondary hover:text-navy transition-colors">Previous</a>`;
    prevBtn.classList.remove('hidden');
  } else {
    prevBtn.classList.add('hidden');
  }

  if (currentPage < totalPages) {
    nextBtn.innerHTML = `<a href="/blog?page=${currentPage + 1}" class="text-sm font-semibold text-text-secondary hover:text-navy transition-colors">Next</a>`;
    nextBtn.classList.remove('hidden');
  } else {
    nextBtn.classList.add('hidden');
  }
}

// Initial render
renderPosts();
